Program 6: Linear Search

Aim

Write a program to search for a given element in an array using Linear Search.

Algorithm

Function Main
    Declare Integer n, i, key
    Declare Boolean found
    Declare Integer Array A[i]

    Input n
    For i = 0 to n-1
        Input A[i]
    End
    Input key
    Assign found = false
    Assign i = 0
    While i<n && found==false
        If A[i]==key
            Assign found = true
            Output "Element found at position" +i
        Else
            Assign i = i+1
        End
    End
    If found==false
        Output "Element not Found"
    End
End

Program 7: Binary Search

Aim

To write a Python program to search for an element in a sorted array using the Binary Search algorithm.

Algorithm

Function Main
    Declare Integer i, n, key, low, high, mid
    Declare Boolean found
    Declare Integer Array arr[100]
    Input n
    For i = 0 to n-1
        Input arr[i]
    End
    Input key
    Assign low = 0
    Assign high = n-1
    Assign found = false
    While low<=high AND found=false
        Assign mid = (low+high)/2
        If arr[mid]==key
            Assign found = true
        Else
            If arr[mid]<key
                Assign low = mid+1
            Else
                Assign high = mid-1
            End
        End
    End
    If found==true
        Output "Element found at position:" &(mid+1)
    Else
        Output "Element not found"
    End
End

Program 11: Quick Sort

Aim

To write a Python program to sort an array in using Quick Sort.

Algorithm

Function Main
    Declare Integer n, i
    Declare Integer Array arr[100]

    Input n
    For i = 0 to n-1
        Input arr[i]
    End
    Call Quicksort(arr, 0, n-1)
    Output "Sorted array:"
    For i = 0 to n-1
        Output arr[i]
    End
End

Function partition (Integer Array arr, Integer low, Integer high)
    Declare Integer pivot, i, j, k, temp

    Assign pivot = arr[high]
    Assign i = low-1
    Assign j = low
    While j<high
        If arr[j]<=pivot
            Assign i = i+1
            Assign temp = arr[i]
            Assign arr[i] = arr[j]
            Assign arr[j] = temp
        Else
            Assign j = j+1
        End
    End
    Assign k = i+1
    Assign temp = arr[k]
    Assign arr[k] = arr[high]
    Assign arr[high] = temp
    Return Integer k
End

Function Quicksort (Integer Array arr, Integer low, Integer high)
    Declare Integer pivot

    If low<high
        Assign pivot = partition(arr, low, high)
        Call Quicksort(arr, low, pivot-1)
        Call Quicksort(arr, pivot+1, high)
    End
End

def quick_sort(arr):
    if len(arr) <= 1:
        return arr
    pivot = arr[0]
    left = [x for x in arr[1:] if x <= pivot]
    right = [x for x in arr[1:] if x > pivot]
    return quick_sort(left) + [pivot] + quick_sort(right)

def merge_sort(arr):
    if len(arr) <= 1:
        return arr
    mid = len(arr) // 2

    right = merge_sort(arr[mid:])

    j = 0

        if left[i] < right[j]:
            result.append(left[i])
            i += 1
        else:
            result.append(right[j])
            j += 1

To write a Python program to traverse a graph using the Breadth First Search (BFS) algorithm.

from collections import deque
class Node:
    def __init__(self, data):
        self.data = data
        self.neighbors = []
class Graph:
    def __init__(self):
        self.nodes = []
    def add_node(self, data):
        node = Node(data)
        self.nodes.append(node)
        return node
    def add_edge(self, node1, node2):
        node1.neighbors.append(node2)
        node2.neighbors.append(node1)
    def print_graph(self):
        for node in self.nodes:
            print(node.data, "->", end=" ")
            for neighbor in node.neighbors:
                print(neighbor.data, end=" ")
            print()
    def BFS(self, start):
        visited = set()
        queue = deque([start])
        visited.add(start)
        while queue:
            node = queue.popleft()
            print(node.data, end=" ")
            for neighbor in node.neighbors:
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append(neighbor)

Program 28: Dijkstra’s Algorithm

Aim

To write a Python program to find the shortest path from a source vertex to all other vertices using Dijkstra’s Algorithm.

Program

class Node:
    def __init__(self, data):
        self.data = data
        self.neighbors = []
class Graph:
    def __init__(self):
        self.nodes = []
    def add_node(self, data):
        node = Node(data)
        self.nodes.append(node)
        return node
    def add_edge(self, node1, node2, weight):
        node1.neighbors.append((node2, weight))
    def print_graph(self):
        for node in self.nodes:
            print(node.data, "->", end=" ")
            for neighbor, weight in node.neighbors:
                print(f"{neighbor.data}({weight})", end=" ")
            print()
    def dijkstra(self, source):
        distance = {}
        for node in self.nodes:
            distance[node] = float('inf')
        distance[source] = 0
        visited = set()
        while len(visited) < len(self.nodes):
            current = None
            min_distance = float('inf')
            for node in self.nodes:
                if node not in visited and distance[node] < min_distance:
                    min_distance = distance[node]
                    current = node
            if current is None:
                break
            visited.add(current)
            for neighbor, weight in current.neighbors:
                new_distance = distance[current] + weight
                if new_distance < distance[neighbor]:
                    distance[neighbor] = new_distance
        return distance
